## 2. GPU并发控制模块

### 2.1 设计思路

GPU显存是稀缺资源，当多个AI推理任务并发执行时，容易发生显存溢出（OOM）。本模块通过信号量（Semaphore）机制控制同时占用GPU的任务数量，确保系统在显存限制内稳定运行。

### 2.2 核心伪代码

```python
# gpu_concurrency.py

class GPUSemaphore:
    """
    GPU显存并发控制器
    - 使用 asyncio.Semaphore 限制并发GPU任务数
    - 防止多请求同时推理导致 OOM
    """
    
    def __init__(self, max_concurrent: int = 1):
        self._semaphore = asyncio.Semaphore(max_concurrent)
        self._device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
        self._active_count = 0
    
    async def __aenter__(self):
        """获取GPU访问权限（阻塞直到可用）"""
        await self._semaphore.acquire()
        self._active_count += 1
        logger.info(f"[GPU] Acquired (active={self._active_count})")
        return self
    
    async def __aexit__(self, exc_type, exc_val, exc_tb):
        """释放GPU访问权限"""
        self._semaphore.release()
        self._active_count -= 1
        logger.info(f"[GPU] Released (active={self._active_count})")
        
        # 清理显存
        if torch.cuda.is_available():
            torch.cuda.empty_cache()
    
    @property
    def available(self) -> int:
        """返回剩余可用槽位"""
        return self._semaphore._value


# 模块级信号量
_gpu_sem = GPUSemaphore(max_concurrent=1)

async def get_gpu_sem() -> GPUSemaphore:
    """获取GPU信号量"""
    return _gpu_sem

async def run_with_gpu_lock(coro):
    """上下文管理器：自动获取/释放GPU锁"""
    async with await get_gpu_sem() as sem:
        result = await coro
        return result
```

## 3. 剧本解析模块

### 3.1 设计思路

剧本解析采用**两阶段流水线**：
1. **标准化阶段（Normalize）**：将任意格式的原始剧本转换为标准格式
2. **解析阶段（Parse）**：将标准化后的剧本解析为结构化数据

解析阶段采用**依赖图并行调用**策略，避免单次LLM调用过长导致超时：
- `角色提取` 和 `元信息提取` 并行执行
- `场景提取` 依赖角色数量
- `段落提取` 依赖场景数量

### 3.4 并行依赖图解析

```python
async def parse_script(normalized_text, language):
    """
    解析阶段 - 依赖图并行调用
    """
    async def _with_retry(coro_factory, attempts=2):
        for i in range(attempts):
            try:
                return await coro_factory()
            except Exception as e:
                if i + 1 < attempts:
                    await asyncio.sleep(0.6 * (i + 1))
        raise

    # Stage 1: chars + header 并行
    header_data, chars = await asyncio.gather(
        _with_retry(lambda: _extract_header(normalized_text, language)),
        _with_retry(lambda: extract_characters(normalized_text, language)),
    )

    # Stage 2: scenes (依赖 chars)
    scenes_list = await _with_retry(
        lambda: _extract_scenes(normalized_text, language)
    )

    # Stage 3: paragraphs (依赖 scene_count)
    scene_count = len(scenes_list) if scenes_list else 1
    paragraphs_data = await _with_retry(
        lambda: _extract_paragraphs(normalized_text, scene_count, language)
    )

    # 组装 ScriptData
    return _assemble_script_data(header_data, chars, scenes_list, paragraphs_data)
```

### 3.5 JSON鲁棒提取

```python
def _extract_json(content: str) -> Optional[dict | list]:
    """
    三级降级策略，从LLM输出中提取JSON
    """
    # Level 1: 直接解析
    try:
        return json.loads(content)
    except json.JSONDecodeError:
        pass

    # Level 2: 去除 markdown 代码块
    fence = re.search(r"```(?:json)?\s*([\s\S]+?)\s*```", content)
    if fence:
        try:
            return json.loads(fence.group(1))
        except json.JSONDecodeError:
            pass

    # Level 3: 贪婪匹配第一个平衡的 { } 或 [ ]
    for opener, closer in [("{", "}"), ("[", "]")]:
        idx = content.find(opener)
        if idx == -1:
            continue
        
        depth = 0
        in_string = False
        escape = False
        
        for end in range(idx, len(content)):
            ch = content[end]
            
            if escape:
                escape = False
                continue
            if ch == "\\":
                escape = True
                continue
            if ch == '"':
                in_string = not in_string
                continue
            if in_string:
                continue
            
            if ch == opener:
                depth += 1
            elif ch == closer:
                depth -= 1
                if depth == 0:
                    try:
                        return json.loads(content[idx:end + 1])
                    except json.JSONDecodeError:
                        break
    
    return None
```

### 3.6 伪角色过滤

```python
# 伪角色黑名单 - 这些词即使出现在冒号后也不是角色
_PSEUDO_CHARACTER_BLACKLIST = frozenset({
    # 镜头/景别词
    "特写", "中景", "近景", "远景", "全景", "俯拍", "仰拍",
    "长镜头", "空镜", "空镜头",
    # 场景元素
    "裂缝", "树林", "漂浮建筑", "地面", "天空", "云", "树叶",
    "灯光", "路灯", "教学楼",
    # 泛指代词
    "她", "他", "他们", "它",
    # 视角/反应
    "她的表情", "周围学生的反应",
})

def _filter_real_characters(characters, paragraphs):
    """
    双重验证确保只保留真实角色：
    1. 不在黑名单中
    2. 在剧本中作为对话说话者或括号描述出现
    """
    # 提取所有对话说话者
    speakers = {p.speaker_id for p in paragraphs if p.speaker_id}
    
    # 提取所有括号描述的角色
    described = set()
    for m in _BRACKET_DESC_RE.finditer(text):  # r"([\u4e00-\u9fffA-Za-z]{2,8})[（(][^)）]+[)）]"
        described.add(m.group(1))
    
    # 提取所有冒号前缀的名字
    for line in text.splitlines():
        m = _DIALOGUE_PREFIX.match(line.strip())  # r"^\s*([A-Z\u4e00-\u9fff]...)\s*[:：]\s*"
        if m:
            described.add(m.group(1).strip())
    
    kept = []
    for c in characters:
        name = c.name
        
        # 剔除黑名单
        if name in _PSEUDO_CHARACTER_BLACKLIST:
            continue
        
        # 长度检查
        if len(name) < 2 or len(name) > 12:
            continue
        
        # 必须作为说话者或括号描述出现
        if name in speakers or name in described:
            kept.append(c)
            continue
        
        # 有元数据(gender/age/personality)但不在剧本中 - 保留
        if c.gender or c.age or c.personality:
            kept.append(c)
    
    return kept
```

### 3.7 段落分类器

```python
def _classify_paragraph(line, speaker_match) -> ParagraphType:
    """
    基于规则+关键词的段落分类
    """
    stripped = line.strip()
    
    # 旁白标记
    if _NARRATION_MARKERS.match(stripped):
        return ParagraphType.NARRATION
    
    # 内心独白标记
    if _INNER_THOUGHT_MARKERS.match(stripped):
        return ParagraphType.INNER
    
    # 转场标记
    for pat in _TRANSITION_PATTERNS:
        if pat.match(stripped):
            return ParagraphType.TRANSITION
    
    # 对话判断
    if speaker_match is not None:
        return ParagraphType.DIALOGUE
    
    # 环境氛围
    if re.match(r"^\s*(?:阳光|月光|灯光|雨|雪|风|雾)\s*[^\n。！？]*$", stripped):
        return ParagraphType.ATMOSPHERE
    
    return ParagraphType.ACTION

def _detect_emotion(text) -> str:
    """
    基于关键词的情绪检测
    """
    _EMOTION_KEYWORDS = {
        "tense": ["紧张", "焦虑", "担心", "恐惧", "害怕"],
        "sad": ["悲伤", "难过", "哭泣", "失望"],
        "joyful": ["开心", "高兴", "幸福", "笑"],
        "angry": ["愤怒", "生气", "恼火"],
        "mysterious": ["神秘", "诡异", "奇怪"],
        "romantic": ["浪漫", "温馨", "甜蜜", "爱"],
    }
    
    low = text.lower()
    for emotion, keywords in _EMOTION_KEYWORDS.items():
        for kw in keywords:
            if kw in low:
                return emotion
    return ""
```

## 4. 图像生成模块

### 4.1 设计思路

图像生成模块采用**Provider Pattern**，支持多后端切换：
- **本地推理**：Z-Image Turbo（蒸馏模型，9步）或 SDXL（25步）
- **云端API**：DashScope 通义万相

关键设计：
1. **Lazy Loading**：首次调用时才加载模型，节省显存
2. **Thread-Safe Singleton**：线程安全的单例模式
3. **自动下载**：Z-Image模型从ModelScope镜像下载

```python
_img_gen: Optional[LocalImageGenerator] = None
_img_gen_lock = threading.Lock()

def get_image_generator() -> LocalImageGenerator:
    """
    线程安全的懒加载单例
    """
    global _img_gen
    if _img_gen is None:
        with _img_gen_lock:  # 双重检查锁定
            if _img_gen is None:
                _img_gen = LocalImageGenerator(
                    checkpoint_dir=_default_checkpoint_dir()

def configure_image_generator(model_id: str, dtype_name: str):
    """
    重新配置时先卸载旧模型，释放GPU显存
    """
    global _img_gen
    with _img_gen_lock:

        _img_gen = LocalImageGenerator(
            model_id=model_id,
            dtype_name=dtype_name,
            checkpoint_dir=_default_checkpoint_dir(),
```

### 4.4 GPU显存管理

```python
def _gpu_vram_gb() -> tuple[float, float]:
    """返回 (free, total) VRAM GB"""
    try:
        free, total = torch.cuda.mem_get_info(0)
        return free / 1024**3, total / 1024**3
    except:
        return 0.0, 0.0

def _clear_cuda_cache_best_effort():
    """
    失败路径清理，避免显存泄漏
    每次错误后调用，确保下次请求有干净的GPU状态
    """
    try:
        if torch.cuda.is_available():
            torch.cuda.empty_cache()
            torch.cuda.synchronize()
        gc.collect()
    except Exception as e:
        logger.warning(f"[ImageGen] cache clear failed: {e}")

def _round_to_multiple(value: int, multiple: int = 16) -> int:
    """Z-Image要求尺寸能被16整除"""
    return max(multiple, ((value + multiple - 1) // multiple) * multiple)
```

```python
def _load_pipeline(model_id, dtype_name, pipeline_type, checkpoint_dir=None):
    """
    模型加载流程：
    1. 诊断快照（记录加载前显存）
    2. 尝试加载主模型
    3. 失败时降级到SDXL
    """
    # 诊断日志
    logger.info(f"[ImageGen] Loading {model_id} (type={pipeline_type})")
    
    try:
        dtype_map = {
            "bfloat16": torch.bfloat16,
            "float16": torch.float16,
            "float32": torch.float32,
        }
        dtype = dtype_map.get(dtype_name, torch.bfloat16)
        device = "cuda" if torch.cuda.is_available() else "cpu"
        
        if pipeline_type == "z_image":
            # Z-Image: 先确保模型文件在本地
            if checkpoint_dir:
                try:
                    from app.models.z_image_loader import ZImageModel
                    loader = ZImageModel(model_id=model_id, checkpoint_dir=checkpoint_dir)
                    load_path = loader.ensure_downloaded()  # 自动从ModelScope下载
                except Exception as e:
                    logger.warning(f"[ImageGen] Local snapshot failed: {e}")
                    load_path = model_id
            else:
                load_path = model_id
            
            # 加载管线
            from diffusers import ZImagePipeline
            pipe = ZImagePipeline.from_pretrained(load_path, torch_dtype=dtype)
        
        else:  # sdxl
            from diffusers import StableDiffusionXLPipeline
            pipe = StableDiffusionXLPipeline.from_pretrained(
                "stabilityai/stable-diffusion-xl-base-1.0",
                torch_dtype=dtype
            )
        
        # 移动到GPU
        pipe.to(device)
        return pipe
    
    except Exception as e:
        logger.warning(f"[ImageGen] Load failed: {e}")
        _clear_cuda_cache_best_effort()
        
        # 降级到SDXL
        try:
            from diffusers import StableDiffusionXLPipeline
            pipe = StableDiffusionXLPipeline.from_pretrained(
                "stabilityai/stable-diffusion-xl-base-1.0",
                torch_dtype=torch.bfloat16
            )
            pipe.to("cuda")
            return pipe
        except:
            return None
```

```python
def predict(self, image: Image.Image, mask: Image.Image) -> Image.Image:
    """
    图像修复
    
    Args:
        image: RGB PIL Image，原始图像
        mask: PIL Image
              - L mode: 255(白色)=待修复区域, 0(黑色)=保留区域
              - RGBA mode: alpha=255=待修复区域, alpha=0=保留区域
    
    Returns:
        修复后的RGB图像
    """
    # 图像预处理
    if image.mode == "RGBA":
        rgb = Image.new("RGB", image.size, (255, 255, 255))
        rgb.paste(image, mask=image.split()[3])
        image = rgb
    elif image.mode != "RGB":
        image = image.convert("RGB")
    
    # Mask预处理
    if isinstance(mask, Image.Image):
        if mask.mode == "RGBA":
            mask = mask.split()[3]  # 提取alpha通道
        elif mask.mode != "L":
            mask = mask.convert("L")
    else:
        mask = Image.fromarray(mask).convert("L")
    
    # 推理
    result = self._model(image, mask)  # Fourier卷积修复
    
    return result
```

```python
def segment_person_from_frame(frame_path, output_path, sam2_model, min_area=1000):
    """
    分割单帧中的人物区域
    
    策略：选择最大连通区域作为人物（假设主要人物占画面主体）
    """
    image = cv2.imread(frame_path)
    if image is None:
        return False
    
    # SAM2自动分割
    masks = sam2_model.predict_automatic_masks(image)
    if not masks:
        return False
    
    # 选择最大连通区域
    person_mask = None
    max_area = 0
    
    for mask_data in masks:
        seg = mask_data.get("segmentation") or mask_data.get("mask")
        if seg is None:
            continue
        
        area = float(seg.sum())
        if area > max_area and area > min_area:
            max_area = area
            person_mask = (seg.astype(np.uint8) * 255)
    
    if person_mask is None:
        return False
    
    # 构建RGBA图像（透明背景）
    h, w = person_mask.shape[:2]
    rgba = np.zeros((h, w, 4), dtype=np.uint8)
    bgr = image[:, :, :3]
    rgb = cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB)
    rgba[:, :, :3] = rgb
    rgba[:, :, 3] = person_mask  # alpha通道
    
    cv2.imwrite(output_path, rgba)
    return True
```

## 7. 时序追踪模块

### 7.1 设计思路

时序追踪模块解决跨帧物体关联问题：同一角色在不同帧中可能出现在不同位置、有不同尺寸。核心挑战是**外观相似度计算**和**贪心二分图匹配**。

特点：
- **非连续帧支持**：适用于分镜生成的关键帧，而非视频连续帧
- **混合匹配策略**：结合语义相似度、外观特征、深度信息、IoU
- **轨迹构建**：记录物体在时间序列中的运动

### 7.3 外观相似度计算

```python
def cosine_similarity(vec1, vec2) -> float:
    """
    计算余弦相似度
    """
    norm1 = np.linalg.norm(vec1)
    norm2 = np.linalg.norm(vec2)
    
    if norm1 < 1e-8 or norm2 < 1e-8:
        return 0.0
    
    return float(np.dot(vec1, vec2) / (norm1 * norm2))

def compute_iou(bbox1, bbox2) -> float:
    """
    计算IoU (Intersection over Union)
    """
    x1 = max(bbox1.x, bbox2.x)
    y1 = max(bbox1.y, bbox2.y)
    x2 = min(bbox1.x + bbox1.w, bbox2.x + bbox2.w)
    y2 = min(bbox1.y + bbox1.h, bbox2.y + bbox2.h)
    
    if x2 <= x1 or y2 <= y1:
        return 0.0
    
    intersection = (x2 - x1) * (y2 - y1)
    area1 = bbox1.w * bbox1.h
    area2 = bbox2.w * bbox2.h
    union = area1 + area2 - intersection
    
    return intersection / union if union > 0 else 0.0

class TemporalTracker:
    def _compute_appearance_similarity(self, obj1, obj2, feat1, feat2) -> float:
        """
        多信号加权外观相似度
        
        权重分配依据各信号可靠性：
        - 类别匹配：最可靠 (35%)
        - 视觉特征：Qwen3-VL伪特征向量 (25%)
        - 深度相似度：20m容忍度 (15%)
        - 尺寸相似度：面积比 (10%)
        - 图层匹配：强信号 (15%)
        """
        # 类别匹配
        class1 = obj1.classLabel.strip().lower()
        class2 = obj2.classLabel.strip().lower()
        class_sim = 1.0 if class1 == class2 else 0.0
        
        # 特征余弦相似度
        feature_sim = cosine_similarity(feat1, feat2)
        
        # 深度相似度 (20m容忍度)
        depth_diff = abs(obj1.depth - obj2.depth)
        depth_sim = max(0, 1.0 - depth_diff / 20.0)
        
        # 尺寸相似度 (面积比)
        area1 = obj1.boundingBox.w * obj1.boundingBox.h
        area2 = obj2.boundingBox.w * obj2.boundingBox.h
        size_ratio = min(area1, area2) / (max(area1, area2) + 1e-8)
        
        # 图层匹配
        layer_sim = 1.0 if obj1.layer == obj2.layer else 0.0
        
        # 加权组合
        combined = (
            0.35 * class_sim +
            0.25 * feature_sim +
            0.15 * depth_sim +
            0.10 * size_ratio +
            0.15 * layer_sim
        )
        
        return combined
```

### 7.4 特征向量构建

```python
def extract_features(self, frame) -> dict[str, np.ndarray]:
    """
    构建64维伪特征向量
    
    组成：
    1. 类别标签哈希 (64维) - 确定性随机
    2. 深度周期编码 (64维) - sin/cos编码
    3. 尺寸编码 (64维) - 面积比、宽高比、中心位置
    
    虽然不是真正的深度特征，但能提供类别+位置+尺寸的区分能力
    """
    features = {}
    
    for obj in frame.objects:
        class_label = obj.classLabel.strip().lower()
        
        # 1. 类别哈希（确定性）
        np.random.seed(hash(class_label) % (2**32))
        base_features = np.random.randn(64)
        base_features = base_features / (np.linalg.norm(base_features) + 1e-8)
        
        # 2. 深度周期编码
        depth_norm = min(obj.depth / 50.0, 1.0)
        depth_encoding = np.array([
            np.sin(depth_norm * np.pi),
            np.cos(depth_norm * np.pi)
        ])
        depth_encoding = np.tile(depth_encoding, 32)[:64]
        
        # 3. 尺寸编码
        area = obj.boundingBox.w * obj.boundingBox.h
        size_encoding = np.array([
            area,
            obj.boundingBox.w / (obj.boundingBox.h + 1e-8),
            obj.boundingBox.x + obj.boundingBox.w / 2,
            obj.boundingBox.y + obj.boundingBox.h / 2,
        ])
        size_encoding = np.tile(size_encoding, 16)[:64]
        
        # 组合
        combined = base_features + 0.3 * depth_encoding + 0.2 * size_encoding
        combined = combined / (np.linalg.norm(combined) + 1e-8)
        
        features[obj.id] = combined
    
    return features
```

### 7.5 贪心二分图匹配

```python
def match_objects(self, current_objs, previous_objs, 
                  current_features, previous_features, threshold=0.6):
    """
    跨帧物体匹配
    
    算法：贪心二分图匹配
    - 按相似度降序遍历候选匹配
    - 每个当前物体最多匹配一个历史物体
    - 每个历史物体最多被一个当前物体匹配
    """
    if not previous_objects:
        return {}
    
    matches = {}
    used_prev = set()
    
    # 构建候选列表
    candidates = []
    for curr_obj in current_objs:
        curr_id = curr_obj.id
        if curr_id not in current_features:
            continue
        
        for prev_obj in previous_objs:
            prev_id = prev_obj.id
            if prev_id not in previous_features:
                continue
            
            score = self._compute_appearance_similarity(
                curr_obj, prev_obj,
                current_features[curr_id],
                previous_features[prev_id]
            )
            
            if score >= threshold:
                candidates.append((score, curr_id, prev_id))
    
    # 按得分降序
    candidates.sort(key=lambda x: -x[0])
    
    # 贪心选择
    for score, curr_id, prev_id in candidates:
        if curr_id not in matches and prev_id not in used_prev:
            matches[curr_id] = prev_id
            used_prev.add(prev_id)
    
    return matches
```

### 7.6 完整追踪流程

```python
def track_sequence(self, frames, matching_threshold=0.6):
    """
    完整跨帧追踪流程
    
    输入: [FrameResult, FrameResult, ...]
    输出: (processed_frames, cross_frame_objects, scene_links)
    """
    processed_frames = []
    cross_frame_objects = {}  # global_id -> CrossFrameObject
    scene_links = []
    
    previous_objects = []
    previous_features = {}
    
    for frame_idx, frame in enumerate(frames):
        # 1. 提取当前帧特征
        current_features = self.extract_features(frame)
        
        # 2. 与前一帧匹配
        if frame_idx > 0 and previous_objects:
            global_id_map = self.match_objects(
                frame.objects,
                previous_objects,
                current_features,
                previous_features,
                threshold=matching_threshold
            )
            
            # 更新全局ID映射
            new_global_ids = {}
            for local_id, prev_local_id in global_id_map.items():
                global_id = self._global_id_map.get(prev_local_id)
                if global_id:
                    new_global_ids[local_id] = global_id
                    frame.globalObjectIds[local_id] = global_id
                    
                    # 添加appearance记录
                    obj = next((o for o in frame.objects if o.id == local_id), None)
                    if obj:
                        cross_frame_objects[global_id].appearances.append(
                            ObjectAppearance(
                                frameId=frame.frameId,
                                frameIndex=frame_idx,
                                localId=local_id,
                                bbox=obj.boundingBox,
                                depth=obj.depth,
                                matchConfidence=0.8,
                            )
                        )
        
        else:
            # 第一帧：分配新全局ID
            for obj in frame.objects:
                global_id = self._get_next_global_id()
                self._global_id_map[obj.id] = global_id
                frame.globalObjectIds[obj.id] = global_id
                
                cross_frame_objects[global_id] = CrossFrameObject(
                    globalId=global_id,
                    classLabel=obj.classLabel,
                    appearances=[
                        ObjectAppearance(
                            frameId=frame.frameId,
                            frameIndex=frame_idx,
                            localId=obj.id,
                            bbox=obj.boundingBox,
                            depth=obj.depth,
                            matchConfidence=1.0,
                        )
                    ],
                )
        
        processed_frames.append(frame)
        
        # 3. 更新状态
        previous_objects = frame.objects
        previous_features = current_features
        
        # 4. 计算场景链接
        if frame_idx > 0:
            link = self._compute_scene_link(frames[frame_idx - 1], frame, cross_frame_objects)
            if link:
                scene_links.append(link)
    
    return processed_frames, list(cross_frame_objects.values()), scene_links
```

```python
def build_trajectory(appearances: list[ObjectAppearance]) -> Trajectory:
    """
    从appearance列表构建运动轨迹
    """
    if not appearances:
        return Trajectory(globalId="", classLabel="")
    
    traj = Trajectory(
        globalId=appearances[0].frameId,
        classLabel="",
        center_x=[],
        center_y=[],
        depth=[],
        area=[],
        frame_indices=[],
    )
    
    for app in appearances:
        bbox = app.bbox
        traj.center_x.append(bbox.x + bbox.w / 2)
        traj.center_y.append(bbox.y + bbox.h / 2)
        traj.depth.append(app.depth)
        traj.area.append(bbox.w * bbox.h)
        traj.frame_indices.append(app.frameIndex)
    
    return traj
```

```python
# 统一的错误处理模式
async def with_retry(coro_factory, label, max_attempts=3):
    """
    指数退避重试
    """
    for attempt in range(max_attempts):
        try:
            return await coro_factory()
        except Exception as e:
            logger.warning(f"[{label}] attempt {attempt+1} failed: {e}")
            if attempt + 1 < max_attempts:
                await asyncio.sleep(0.5 * (2 ** attempt))  # 指数退避
            else:
                logger.error(f"[{label}] exhausted retries")
                raise

def best_effort_cleanup():
    """
    最佳努力清理
    任何清理失败都不应抛出异常
    """
    try:
        torch.cuda.empty_cache()
        gc.collect()
    except:
        pass  # 静默忽略
```
